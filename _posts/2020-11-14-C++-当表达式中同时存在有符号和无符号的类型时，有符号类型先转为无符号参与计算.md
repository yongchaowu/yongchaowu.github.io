```cpp
    /*当表达式中同时存在有符号和无符号的类型时，有符号类型先转为无符号参与计算*/
    unsigned int un1 = 1;
    int n2 = -1;
    std::cout<< "1[unsigned int] * -1[int]" <<un1 * n2<<std::endl; //4294967295
```
